KEY_LBA=34

cryptsetup open --type plain \

        --key-size 512 \
        --key-file "$DISK" \
        --keyfile-offset $(( KEY_LBA * 512 )) \
        --keyfile-size 64 \
        --sector-size 512 \
        --skip 100352 \
        --offset 100352 \
